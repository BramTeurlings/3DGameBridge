cmake_minimum_required(VERSION 3.23)

set (CMAKE_CXX_STANDARD 20)

project (GameBridge VERSION 0.1)
message("Game Bridge Version: ${3DGameBridge_VERSION}")

# Todo Don't really like that we add a submodule here for unit tests, in gamebridge/unit_tests. Maybe use CPM for this?
# Option to build unit tests project
option(BUILD_UNIT_TESTS "Enabled the unit_tests project")
if(BUILD_UNIT_TESTS)
	add_subdirectory(third-party/googletest)
endif()

#  link with LeiaSR SDK
if(DEFINED LEIASR_SDKROOT)
	message("LeiaSR SDK found in ${LEIASR_SDKROOT}")
elseif(DEFINED ENV{LEIASR_SDKROOT})
	set(LEIASR_SDKROOT $ENV{LEIASR_SDKROOT})
	message("LeiaSR SDK found with environment variable ${LEIASR_SDKROOT}")
else()
	message(FATAL_ERROR "LEIASR_SDKROOT not found, either set it or add it as an environment variable.")
endif()

list(APPEND CMAKE_PREFIX_PATH "${LEIASR_SDKROOT}/lib/cmake")


# Include SR
find_package(simulatedreality REQUIRED)
find_package(srDirectX REQUIRED)

# Other dependencies
add_subdirectory(third-party/zydis EXCLUDE_FROM_ALL)
add_subdirectory(third-party/safetyhook EXCLUDE_FROM_ALL)

# Add projects
add_subdirectory(gamebridge)
